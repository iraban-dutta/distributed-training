# ================================================================
# Data
# ================================================================

# Download and tokenize TinyStories
python -m main.download_preprocess


# ================================================================
# Train-step throughput
# ================================================================
# Prints per-step tokens/sec, then the mean over the timed steps.
# Save each run under a distinct name so the numbers can be compared later.

mkdir -p logs/tput

# Dense
PYTHONUNBUFFERED=1 python -m main.train_benchmark \
  --config config.dense_default \
  --batch-size 8 \
  | tee logs/tput/dense_b8.txt

# Dense, other batch sizes
PYTHONUNBUFFERED=1 python -m main.train_benchmark \
  --config config.dense_default \
  --batch-size 16 \
  | tee logs/tput/dense_b16.txt

# MoE (4 experts, top-1, capacity 1.25, vectorized dispatch)
PYTHONUNBUFFERED=1 python -m main.train_benchmark \
  --config config.moe_default \
  --batch-size 8 \
  | tee logs/tput/moe_b8.txt

PYTHONUNBUFFERED=1 python -m main.train_benchmark \
  --config config.moe_default \
  --batch-size 16 \
  | tee logs/tput/moe_b16.txt


# ================================================================
# Dense vs MoE breakdown
# ================================================================
# Times data / forward / backward / optimizer and prints tok/s for each.
# MoE runs also split the forward into route, dispatch, expert GEMM, combine.
#
# Set these at the top of main/train_benchmark_detailed.py before each run:
#   USE_MOE                  False = dense, True = MoE
#   USE_VECTORIZED_DISPATCH  MoE only. True = vectorized, False = legacy
#   TOPK                     1 or 2. TOPK=2 halves the batch so token count matches dense
#   CAPACITY_FACTOR          2 or 4
#   WARMUP_STEPS, TRAIN_STEPS

mkdir -p logs/tput_detailed

# Example: dense
# USE_MOE = False
PYTHONUNBUFFERED=1 python -m main.train_benchmark_detailed \
  | tee logs/tput_detailed/dense.txt

# Example: MoE, vectorized dispatch, top-1, capacity 4
# USE_MOE = True
# USE_VECTORIZED_DISPATCH = True
# TOPK = 1
# CAPACITY_FACTOR = 4
PYTHONUNBUFFERED=1 python -m main.train_benchmark_detailed \
  | tee logs/tput_detailed/moe_vectorized_k1_cf4.txt

# Example: MoE, legacy dispatch, same routing settings
# USE_VECTORIZED_DISPATCH = False
PYTHONUNBUFFERED=1 python -m main.train_benchmark_detailed \
  | tee logs/tput_detailed/moe_legacy_k1_cf4.txt

# Background run
PYTHONUNBUFFERED=1 nohup python -m main.train_benchmark_detailed \
  > logs/tput_detailed/moe_vectorized_k1_cf4.txt 2>&1 &
tail -f logs/tput_detailed/moe_vectorized_k1_cf4.txt


# ================================================================
# Training
# ================================================================
# The trainer log line includes TPUT (tok/sec) for each logged step.
# Written to logs/<run>/train.log as well as stdout.

python -m main.train --help

python -m main.train --config config.dense_default

python -m main.train --config config.moe_default

python -m main.train --config config.dense_default \
  --batch-size 16 \
  --lr 6e-4 \
  --log-interval 10 \
  --eval-interval 200 \
  --checkpoint-interval 1000


# ================================================================
# Resume
# ================================================================
# Model config comes from the checkpoint. Trainer settings can still be overridden.

python -m main.train --resume checkpoints/<run>/latest.pt \
  --log-interval 10 \
  --eval-interval 100 \
  --checkpoint-interval 1000


# ================================================================
# Inference throughput (cached vs naive)
# ================================================================

python -m main.infer --help

python -m main.infer --config config.dense_default --prompt "Once upon a time"

python -m main.infer \
  --ckpt checkpoints/<run>/best.pt \
  --prompt "Once upon a time" \
  --num-samples 5 \
  --max-new-tokens 50 \
  --strategy topk \
  --temperature 1.0 \
  --top-k 50

python -m main.infer \
  --ckpt checkpoints/<run>/best.pt \
  --benchmark \
  | tee logs/tput/infer_cached_vs_naive.txt
